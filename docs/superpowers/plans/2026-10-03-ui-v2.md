```rust
/// Format a running total: `m:ss`, `h:mm:ss` from one hour, `d:hh:mm:ss` from one day.
pub fn fmt_total(d: Duration) -> String {
    let total = d.as_secs();
    let (days, rest) = (total / 86_400, total % 86_400);
    if days == 0 {
        return fmt_duration(Duration::from_secs(rest));
    }
    let (h, m, s) = (rest / 3600, (rest % 3600) / 60, rest % 60);
    format!("{days}:{h:02}:{m:02}:{s:02}")
}

/// Status-bar text for a list of `count` tracks lasting `total`.
pub fn fmt_status(count: usize, total: Duration) -> String {
    let noun = if count == 1 { "song" } else { "songs" };
    format!("{count} {noun}, {}", fmt_total(total))
}
```

```rust
//! Play order and cursor. Pure data, no I/O.

use crate::track::TrackId;

/// What happens when a track or the whole queue ends.
#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
pub enum Repeat {
    /// Stop after the last track.
    #[default]
    Off,
    /// Start over after the last track.
    All,
    /// Replay the current track when it ends; manual next/previous wrap like `All`.
    One,
}

/// The ordered list of tracks that next/previous move through.
///
/// `order` is what the user sees (the table). `play_order` is what playback
/// walks: identical to `order` unless shuffle is on.
#[derive(Debug, Clone)]
pub struct Queue {
    order: Vec<TrackId>,
    play_order: Vec<TrackId>,
    /// Index into `play_order`.
    current: Option<usize>,
    shuffle: bool,
    repeat: Repeat,
    rng: fastrand::Rng,
}

impl Default for Queue {
    fn default() -> Self {
        Self::with_rng(fastrand::Rng::new())
    }
}

impl Queue {
    /// An empty queue with shuffle and repeat off.
    pub fn new() -> Self {
        Self::default()
    }

    /// An empty queue whose shuffles are reproducible.
    pub fn with_seed(seed: u64) -> Self {
        Self::with_rng(fastrand::Rng::with_seed(seed))
    }

    fn with_rng(rng: fastrand::Rng) -> Self {
        Self {
            order: Vec::new(),
            play_order: Vec::new(),
            current: None,
            shuffle: false,
            repeat: Repeat::Off,
            rng,
        }
    }

    /// Replace the order. The current track is kept if it is still present;
    /// with shuffle on the new order is reshuffled with it first.
    pub fn set_order(&mut self, order: Vec<TrackId>) {
        let current = self.current();
        self.order = order;
        self.rebuild(current);
    }

    /// The visible order, unaffected by shuffle.
    pub fn order(&self) -> &[TrackId] {
        &self.order
    }

    /// The track the cursor is on.
    pub fn current(&self) -> Option<TrackId> {
        self.current.and_then(|i| self.play_order.get(i).copied())
    }

    /// Whether shuffle is on.
    pub fn shuffle(&self) -> bool {
        self.shuffle
    }

    /// Turn shuffle on or off. The cursor stays on the current track.
    pub fn set_shuffle(&mut self, shuffle: bool) {
        if self.shuffle == shuffle {
            return;
        }
        let current = self.current();
        self.shuffle = shuffle;
        self.rebuild(current);
    }

    /// The repeat mode.
    pub fn repeat(&self) -> Repeat {
        self.repeat
    }

    /// Set the repeat mode.
    pub fn set_repeat(&mut self, repeat: Repeat) {
        self.repeat = repeat;
    }

    /// Move the cursor to `id` without changing the play order.
    /// Returns `false` if it is not in the queue.
    pub fn jump_to(&mut self, id: TrackId) -> bool {
        match self.position_of(id) {
            Some(i) => {
                self.current = Some(i);
                true
            }
            None => false,
        }
    }

    /// Begin playing from `id`, chosen by the user from the visible list.
    /// With shuffle on, the rest of the queue is reshuffled to follow it.
    /// Returns `false` if it is not in the queue.
    pub fn start_at(&mut self, id: TrackId) -> bool {
        if !self.order.contains(&id) {
            return false;
        }
        if self.shuffle {
            self.rebuild(Some(id));
            true
        } else {
            self.jump_to(id)
        }
    }

    /// Advance. Starts at the front when there is no cursor. At the end,
    /// returns `None` under [`Repeat::Off`] and wraps otherwise (reshuffling
    /// first when shuffle is on).
    // Not an iterator: the queue is navigated in both directions.
    #[allow(clippy::should_implement_trait)]
    pub fn next(&mut self) -> Option<TrackId> {
        let len = self.play_order.len();
        if len == 0 {
            return None;
        }
        let next = match self.current {
            None => 0,
            Some(i) if i + 1 < len => i + 1,
            Some(_) if self.repeat == Repeat::Off => return None,
            Some(_) => {
                if self.shuffle {
                    self.rng.shuffle(&mut self.play_order);
                }
                0
            }
        };
        self.current = Some(next);
        self.current()
    }

    /// Step back. Starts at the end when there is no cursor. At the front,
    /// returns `None` under [`Repeat::Off`] and wraps otherwise.
    pub fn prev(&mut self) -> Option<TrackId> {
        let len = self.play_order.len();
        if len == 0 {
            return None;
        }
        let prev = match self.current {
            None => len - 1,
            Some(0) if self.repeat == Repeat::Off => return None,
            Some(0) => len - 1,
            Some(i) => i - 1,
        };
        self.current = Some(prev);
        self.current()
    }

    /// What to play when the current track ends on its own:
    /// the same track under [`Repeat::One`], otherwise [`Self::next`].
    pub fn advance(&mut self) -> Option<TrackId> {
        if self.repeat == Repeat::One && self.current.is_some() {
            return self.current();
        }
        self.next()
    }

    /// Up to `n` tracks that successive [`Self::next`] calls would yield.
    ///
    /// In plain order under `All`/`One` the list continues around to the track
    /// before the current one. With shuffle on it ends at the end of the
    /// current pass, because the next pass is reshuffled when it starts.
    pub fn upcoming(&self, n: usize) -> Vec<TrackId> {
        let len = self.play_order.len();
        let start = self.current.map_or(0, |i| i + 1);
        let wraps = self.repeat != Repeat::Off && !self.shuffle;
        let available = if wraps {
            len - usize::from(self.current.is_some() && len > 0)
        } else {
            len.saturating_sub(start)
        };
        (0..available.min(n))
            .map(|k| self.play_order[(start + k) % len])
            .collect()
    }

    /// Recompute `play_order` from `order`, putting `first` at the front when
    /// shuffling, and point the cursor at `first`.
    fn rebuild(&mut self, first: Option<TrackId>) {
        self.play_order.clone_from(&self.order);
        if self.shuffle {
            self.rng.shuffle(&mut self.play_order);
            if let Some(pos) = first.and_then(|id| self.position_of(id)) {
                self.play_order.swap(0, pos);
            }
        }
        self.current = first.and_then(|id| self.position_of(id));
    }

    fn position_of(&self, id: TrackId) -> Option<usize> {
        self.play_order.iter().position(|t| *t == id)
    }
}
```

```rust
    for track in tracks {

    unique_sorted(tracks.iter().map(|t| t.album.as_str()))
        .into_iter()

                track_count: members.len(),
```

```rust
/// Decode `bytes` and shrink so neither edge exceeds `max_edge`.
/// Returns `None` when the bytes are not a supported image.
pub fn thumbnail(bytes: &[u8], max_edge: u32) -> Option<Arc<RenderImage>> {
    let decoded = image::load_from_memory(bytes).ok()?;
    let small = if decoded.width() > max_edge || decoded.height() > max_edge {
        decoded.thumbnail(max_edge, max_edge)
    } else {
        decoded
    };
    let mut pixels = small.into_rgba8();
    // gpui paints BGRA.
    for pixel in pixels.chunks_exact_mut(4) {
        pixel.swap(0, 2);
    }
    Some(Arc::new(RenderImage::new(vec![Frame::new(pixels)])))
}
```

```rust
//! Cover-art thumbnails, read and decoded off the UI thread.

use std::collections::HashMap;
use std::sync::Arc;

#[allow(clippy::wildcard_imports)] // gpui_kit re-exports the gpui prelude
use gpui_kit::*;
use tunes_core::{Library, TrackId};

use crate::ui::art;

/// Longest edge of a cached thumbnail, in pixels.
const THUMB_EDGE: u32 = 320;

/// What the cache knows about one track's art.
#[derive(Clone)]
pub enum ArtState {
    /// Being read; observers are notified when it resolves.
    Loading,
    /// Decoded thumbnail.
    Ready(Arc<RenderImage>),
    /// No art, or it could not be decoded.
    Missing,
}

/// Shared thumbnail cache. Observe the entity to re-render when art arrives.
pub struct ArtCache {
    library: Option<Arc<Library>>,
    entries: HashMap<TrackId, ArtState>,
}

impl ArtCache {
    /// Empty cache with no library.
    pub fn new(_cx: &mut Context<Self>) -> Self {
        Self {
            library: None,
            entries: HashMap::new(),
        }
    }

    /// Install the library the ids refer to and forget everything cached.
    pub fn set_library(&mut self, library: Option<Arc<Library>>, cx: &mut Context<Self>) {
        self.library = library;
        self.entries.clear();
        cx.notify();
    }

    /// The thumbnail for `id`. A first request returns `Loading` and starts
    /// one background read; later requests return the stored state.
    pub fn get(&mut self, id: TrackId, cx: &mut Context<Self>) -> ArtState {
        if let Some(state) = self.entries.get(&id) {
            return state.clone();
        }
        let Some(library) = self.library.clone() else {
            return ArtState::Missing;
        };
        if !library.track(id).is_some_and(|t| t.has_art) {
            self.entries.insert(id, ArtState::Missing);
            return ArtState::Missing;
        }
        self.entries.insert(id, ArtState::Loading);
        cx.spawn(async move |this, cx| {
            let image = cx
                .background_executor()
                .spawn(async move {
                    let bytes = library.cover_art(id)?;
                    let image = art::thumbnail(&bytes, THUMB_EDGE);
                    if image.is_none() {
                        tracing::warn!(?id, "cover art could not be decoded");
                    }
                    image
                })
                .await;
            this.update(cx, |this, cx| {
                this.entries
                    .insert(id, image.map_or(ArtState::Missing, ArtState::Ready));
                cx.notify();
            })
            .ok();
        })
        .detach();
        ArtState::Loading
    }
}
```

```rust
/// Columns that fit in `width`, never fewer than one.
fn columns_for(width: f32) -> usize {
    // Padding of one GAP on each side; n tiles need n*TILE + (n-1)*GAP.
    let usable = width - GAP;
    #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
    let fit = (usable / (TILE + GAP)).floor().max(1.) as usize;
    fit
}

            selected: None,
            width: px(0.),
            _subscription: subscription,

impl Render for AlbumGrid {
    fn render(&mut self, _: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
        let album_count = self
            .library
            .as_ref()
            .map_or(0, |l| l.album_summaries().len());

        let row_count = album_count.div_ceil(columns);

            .border_b_1()

                        });

                .size_full()
                .pt(px(GAP)),
            )
    }
```
